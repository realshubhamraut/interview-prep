65.8 Health Check Endpoints

@app.route("/health")
def health_check():
    checks = {
        "database": check_postgres(),
        "redis": check_redis(),
        "qdrant": check_qdrant(),
        "kafka": check_kafka(),
        "disk_space": check_disk(),
        "memory": check_memory()
    }
    
    status = "healthy" if all(checks.values()) else "degraded"
    
    return jsonify({
        "status": status,
        "checks": checks,
        "uptime": get_uptime(),
        "version": app.config.get("VERSION", "1.0.0")
    }), 200 if status == "healthy" else 503
